# V0.2 Engineering Skeleton Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the first V0.2 engineering skeleton so BridgeAI-Agent has testable Tool SDK, Workflow, Agent Runner, backend health, frontend placeholder, migration placeholder, and local development instructions.

**Architecture:** Keep the first implementation deliberately small and dependency-light. Core behavior lives in pure Python modules that can be tested without network installs; FastAPI and Vue/Vite entry files are provided as runnable skeletons once dependencies are installed.

**Tech Stack:** Python 3.12+, pytest, optional FastAPI backend entry, Vue 3 + TypeScript + Vite frontend skeleton, PostgreSQL migration placeholder.

## Global Constraints

- Work on branch `codex/v0.2-engineering-skeleton`.
- Use TDD for production Python behavior: write failing tests first, verify red, implement minimal code, verify green.
- Do not install network dependencies during this first pass unless the user approves.
- Do not commit secrets, credentials, real project data, model weights, or generated large artifacts.
- Keep `temp/` absent; do not recreate historical reference directories.
- Keep V0.2 scope to engineering skeleton and local run instructions; do not implement real disease detection models.

def test_executor_invokes_registered_tool_and_records_manifest_version():
    registry = ToolRegistry()
    manifest = ToolManifest(
        tool_id="image_quality_check",
        version="0.1.0",
        name="Image quality check",
        input_schema={"required": ["artifact_id"]},
        output_schema={"required": ["quality_status"]},
    )
    registry.register(manifest, lambda payload: {"quality_status": "pass", "artifact_id": payload["artifact_id"]})

    result = ToolExecutor(registry).execute("image_quality_check", {"artifact_id": "art_001"})

    assert result.ok is True
    assert result.tool_id == "image_quality_check"
    assert result.version == "0.1.0"
    assert result.output == {"quality_status": "pass", "artifact_id": "art_001"}


def test_executor_rejects_payload_missing_required_input():
    registry = ToolRegistry()
    registry.register(
        ToolManifest(
            tool_id="image_quality_check",
            version="0.1.0",
            name="Image quality check",
            input_schema={"required": ["artifact_id"]},
            output_schema={"required": ["quality_status"]},
        ),
        lambda payload: {"quality_status": "pass"},
    )

    result = ToolExecutor(registry).execute("image_quality_check", {})

    assert result.ok is False
    assert result.error_code == "missing_required_input"
    assert result.output == {}

    assert next_state.status == WorkflowStatus.RUNNING
    assert next_state.current_step == "data_check"
    assert next_state.history[-1].step_name == "data_check"
    assert next_state.history[-1].output == {"artifact_id": "art_001"}


def test_workflow_can_fail_and_recover_to_previous_step():
    state = WorkflowState.create(task_id="task_001").advance("data_check", {"artifact_id": "art_001"})

    failed = state.fail("tool_execution", "tool timeout")
    recovered = failed.recover("data_check")

    assert failed.status == WorkflowStatus.FAILED
    assert failed.error_message == "tool timeout"
    assert recovered.status == WorkflowStatus.RUNNING
    assert recovered.current_step == "data_check"
    assert recovered.error_message is None
